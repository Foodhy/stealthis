import { LumenClient } from "@lumen/sdk";

const client = new LumenClient({ lane: "payments" });

const tx = await client.transfer({
  to: "0x7a3f…c41d",
  amount: 12.5,        // LMN
  memo: "gm",
});

console.log(tx.hash);  // finalized in ~380ms
use lumen_sdk::{Client, Lane};

#[tokio::main]
async fn main() -> anyhow::Result<()> {
    let client = Client::new(Lane::Payments).await?;

    let tx = client
        .transfer("0x7a3f…c41d", 12.5)
        .memo("gm")
        .send()
        .await?;

    println!("finalized: {}", tx.hash);
    Ok(())
}
# install the toolchain
curl -sSf https://get.lumen.example | sh

# spin up a localnet with instant blocks
lumen dev --fast

# send 12.5 LMN on the payments lane
lumen tx transfer 0x7a3f…c41d 12.5 \
  --lane payments --memo "gm"
