        input.setAttribute("aria-label", "Digit " + (i + 1));
        input.setAttribute("autocomplete", i === 0 ? "one-time-code" : "off");

        group.appendChild(input);

            if (box.value) {

              boxes[i - 1].focus();
              boxes[i - 1].value = "";

            return;

            boxes[i - 1].focus();
            e.preventDefault();

            boxes[i + 1].focus();
            e.preventDefault();
            return;

          if (e.key.length === 1 && !/[0-9]/.test(e.key) && !e.ctrlKey && !e.metaKey) {
            e.preventDefault();
